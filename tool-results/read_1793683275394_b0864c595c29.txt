    48→async function api<T = unknown>(path: string, method: string, body?: unknown): Promise<T> {
    49→  const res = await fetch(path, {
    50→    method,
    51→    headers: { 'Content-Type': 'application/json' },
    52→    body: body ? JSON.stringify(body) : undefined,
    53→  })
    54→  if (!res.ok) {
    55→    const err = await res.json().catch(() => ({}))
    56→    throw new Error((err as { error?: string }).error || 'Terjadi kesalahan')
    57→  }
    58→  return res.json() as Promise<T>
    59→}

    67→/** Rute simulasi LOOP di sekitar posisi pengguna (bukan rute tetap). */
    68→function makeSimRoute(center: LatLng): [number, number][] {
    69→  const route: [number, number][] = []
    70→  const points = 16
    71→  const radii = [0.42, 0.48, 0.4, 0.34, 0.3, 0.32, 0.38, 0.46, 0.5, 0.42, 0.36, 0.3, 0.28, 0.34, 0.4, 0.46]
    72→  for (let i = 0; i < points; i++) {
    73→    const ang = (i / points) * Math.PI * 2
    74→    const radKm = radii[i]
    75→    const dLat = (radKm / 111.32) * Math.cos(ang)
    76→    const dLng =
    77→      (radKm / (111.32 * Math.max(0.2, Math.cos((center.lat * Math.PI) / 180)))) * Math.sin(ang)
    78→    route.push([center.lat + dLat, center.lng + dLng])
    79→  }
    80→  route.push([route[0][0], route[0][1]])
    81→  return route
    82→}

    84→function advanceSim(route: [number, number][], sp: SimState, stepKm: number): LatLng {
    85→  let from = route[sp.seg]
    86→  let to = route[(sp.seg + 1) % route.length]
    87→  let segLen = Math.max(0.02, haversineKm({ lat: from[0], lng: from[1] }, { lat: to[0], lng: to[1] }))
    88→  let remain = stepKm
    89→  while (remain > 0) {
    90→    const remainSeg = (1 - sp.frac) * segLen
    91→    if (remain < remainSeg) {
    92→      sp.frac += remain / segLen
    93→      remain = 0
    94→    } else {
    95→      remain -= remainSeg
    96→      sp.seg = (sp.seg + 1) % route.length
    97→      sp.frac = 0
    98→      from = route[sp.seg]
    99→      to = route[(sp.seg + 1) % route.length]
   100→      segLen = Math.max(0.02, haversineKm({ lat: from[0], lng: from[1] }, { lat: to[0], lng: to[1] }))
   101→    }
   102→  }
   103→  return { lat: from[0] + (to[0] - from[0]) * sp.frac, lng: from[1] + (to[1] - from[1]) * sp.frac }
   104→}

   109→function greeting(): string {
   110→  const h = new Date().getHours()
   111→  if (h >= 4 && h < 11) return 'Selamat pagi'
   112→  if (h >= 11 && h < 15) return 'Selamat siang'
   113→  if (h >= 15 && h < 18) return 'Selamat sore'
   114→  return 'Selamat malam'
   115→}

   117→function todayInput(): string {
   118→  const d = new Date()
   119→  const pad = (v: number) => String(v).padStart(2, '0')
   120→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
   121→}

  1059→function VehicleFormSheet({
  1060→  open,
  1061→  onClose,
  1062→  vehicle,
  1063→  onSubmit,
  1064→  onDelete,
  1065→}: {
  1066→  open: boolean
  1067→  onClose: () => void
  1068→  vehicle?: VehicleDTO
  1069→  onSubmit: (data: { name: string; type: string; plate?: string; odometer: number }) => Promise<void>
  1070→  onDelete?: () => void
  1071→}) {
  1072→  const [name, setName] = useState('')
  1073→  const [type, setType] = useState('motor')
  1074→  const [plate, setPlate] = useState('')
  1075→  const [odometer, setOdometer] = useState('')
  1076→  const [saving, setSaving] = useState(false)
  1077→
  1078→  useEffect(() => {
  1079→    if (open) {
  1080→      setName(vehicle?.name ?? '')
  1081→      setType(vehicle?.type ?? 'motor')
  1082→      setPlate(vehicle?.plate ?? '')
  1083→      setOdometer(vehicle ? String(vehicle.odometer) : '')
  1084→    }
  1085→  }, [open, vehicle])
  1086→
  1087→  const submit = async () => {
